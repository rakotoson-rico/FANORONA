def rectangle_arrondi(x1, y1, x2, y2, r, **options):
    # polygone "lissé" (smooth=True) = rectangle aux coins arrondis
    points = [
        x1 + r, y1,  x2 - r, y1,  x2, y1,  x2, y1 + r,
        x2, y2 - r,  x2, y2,  x2 - r, y2,  x1 + r, y2,
        x1, y2,  x1, y2 - r,  x1, y1 + r,  x1, y1,
    ]
    return canvas.create_polygon(points, smooth=True, **options)


def dessiner_fond(largeur, hauteur):
    global image_jeu, taille_image_jeu
    # on ne redimensionne que si la fenêtre a changé de taille (sinon c'est lent à chaque clic)
    if taille_image_jeu != (largeur, hauteur):
        ratio = max(
            largeur / image_jeu_originale.width, hauteur / image_jeu_originale.height
        )  # même logique "cover" que ton menu
        image = image_jeu_originale.resize(
            (int(image_jeu_originale.width * ratio), int(image_jeu_originale.height * ratio)),
            Image.Resampling.LANCZOS,
        )
        image_jeu = ImageTk.PhotoImage(image)
        taille_image_jeu = (largeur, hauteur)
    canvas.create_image(largeur / 2, hauteur / 2, image=image_jeu)


def dessiner_bouton(tag, texte, cx, cy, commande, largeur=170, hauteur=52):
    x1, y1 = cx - largeur / 2, cy - hauteur / 2   # coin haut-gauche
    x2, y2 = cx + largeur / 2, cy + hauteur / 2   # coin bas-droit
    zones_boutons.append((x1, y1, x2, y2))        # mémorisé pour clic_case
    tag_corps = tag + "_corps"                    # tag du fond, pour l'effet survol

    # ombre portée (décalée en bas à droite)
    rectangle_arrondi(x1 + 3, y1 + 5, x2 + 3, y2 + 5, 14, fill="#2a1809", outline="", tags=(tag,))
    # corps du bouton : bois foncé + contour doré
    rectangle_arrondi(x1, y1, x2, y2, 14, fill="#5a3319", outline="#c9a24a", width=2, tags=(tag, tag_corps))
    # liseré intérieur (effet bois sculpté)
    rectangle_arrondi(x1 + 5, y1 + 5, x2 - 5, y2 - 5, 10, fill="", outline="#8f5a2e", width=1, tags=(tag,))
    # texte : une copie sombre décalée (ombre) puis le texte crème par-dessus
    canvas.create_text(cx + 1, cy + 2, text=texte, font=("Georgia", 15, "bold"), fill="#1c0f05", tags=(tag,))
    canvas.create_text(cx, cy, text=texte, font=("Georgia", 15, "bold"), fill="#f3e2b8", tags=(tag,))

    # clic + effet survol (le bois s'éclaircit, le curseur devient une main)
    canvas.tag_bind(tag, "<Button-1>", lambda event: commande())
    canvas.tag_bind(tag, "<Enter>", lambda event: (canvas.itemconfig(tag_corps, fill="#7a4a26"), canvas.config(cursor="hand2")))
    canvas.tag_bind(tag, "<Leave>", lambda event: (canvas.itemconfig(tag_corps, fill="#5a3319"), canvas.config(cursor="")))

4. lancer() : version simplifiée

Plus besoin de ttk.Button ni de styles :

python
def lancer():
    for widget in root.winfo_children():
        widget.destroy()
    global canvas, image_jeu_originale, taille_image_jeu
    canvas = tk.Canvas(root, bg="#2b1a0e", highlightthickness=0)
    canvas.pack(fill="both", expand=True)

    image_jeu_originale = Image.open("grille.jpg")   # chargée une seule fois
    taille_image_jeu = None                          # force le premier redimensionnement

    canvas.bind("<Configure>", dessin_plateau)
    canvas.bind("<Button-1>", clic_case)

    nouvelle_partie()
    dessin_plateau()
